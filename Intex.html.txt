  <script>
    var TELEGRAM_BOT_TOKEN = '8727889570:AAEEWSh5TrH7QTLAiSQ_goGREysKmYntcqs'; 
    var TELEGRAM_CHAT_ID = '5707806340';

    var currentSlide = 0;
    var totalSlides = 4;
    var autoSlideInterval;

    function goToSlide(index) {
      currentSlide = index;
      var wrapper = document.getElementById('slidesWrapper');
      wrapper.style.left = '-' + (currentSlide * 100) + '%';
      
      var dots = document.querySelectorAll('.dot');
      for (var i = 0; i < dots.length; i++) {
        if (i === currentSlide) {
          dots[i].className = 'dot active';
        } else {
          dots[i].className = 'dot';
        }
      }

      resetAutoSlide();
    }

    function nextSlide() {
      currentSlide = (currentSlide + 1) % totalSlides;
      goToSlide(currentSlide);
    }

    function startAutoSlide() {
      autoSlideInterval = setInterval(nextSlide, 7000);
    }

    function resetAutoSlide() {
      clearInterval(autoSlideInterval);
      startAutoSlide();
    }

    window.onload = function() {
      var query = window.location.search.substring(1);
      var vars = query.split('&');
      for (var i = 0; i < vars.length; i++) {
        var pair = vars[i].split('=');
        if (pair[0] === 'name') {
          var name = decodeURIComponent(pair[1]);
          document.getElementById('guestNameDisplay').innerHTML = name;
          document.getElementById('wishSenderName').value = name;
        }
      }
    };

    function startExperience() {
      var overlay = document.getElementById('welcome-overlay');
      overlay.style.display = 'none';

      var audio = document.getElementById('bgMusic');
      try {
        audio.play();
      } catch (e) {}

      startAutoSlide();
    }

    function toggleMusic() {
      var audio = document.getElementById('bgMusic');
      if (audio.paused) {
        audio.play();
      } else {
        audio.pause();
      }
    }

    function sendToTelegram(text) {
      var url = 'https://api.telegram.org/bot' + TELEGRAM_BOT_TOKEN + '/sendMessage';
      var xhr = new XMLHttpRequest();
      xhr.open('POST', url, true);
      xhr.setRequestHeader('Content-Type', 'application/json');
      xhr.send(JSON.stringify({
        chat_id: TELEGRAM_CHAT_ID,
        text: text,
        parse_mode: 'HTML'
      }));
    }

    function sendRSVP(status) {
      var guestName = document.getElementById('guestNameDisplay').innerHTML;
      var text = "<b>💌 TO'YGA TASHRIF JAVOBI</b>\n\n<b>Mehmon:</b> " + guestName + "\n<b>Javob:</b> " + status;
      sendToTelegram(text);
      alert("Rahmat! Javobingiz yetkazildi.");
    }

    function sendWish(e) {
      if (e.preventDefault) e.preventDefault();
      var sender = document.getElementById('wishSenderName').value;
      var msg = document.getElementById('wishMessage').value;
      var statusMsg = document.getElementById('statusMsg');

      var text = "<b>🎉 YANGI TABRIKNOMA!</b>\n\n<b>Yuboruvchi:</b> " + sender + "\n<b>Tabrik:</b> " + msg + "\n\n<i>Kamronbek & Nmalumxon Nikoh To'yi</i>";

      statusMsg.innerHTML = "Yuborilmoqda...";
      sendToTelegram(text);
      setTimeout(function() {
        statusMsg.innerHTML = "Ezguli tilagingiz yetkazildi! Rahmat 💖";
        document.getElementById('wishMessage').value = '';
      }, 1000);

      return false;
    }
