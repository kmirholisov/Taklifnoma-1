  <script>
    const TELEGRAM_BOT_TOKEN = '8727889570:AAEEWSh5TrH7QTLAiSQ_goGREysKmYntcqs'; 
    const TELEGRAM_CHAT_ID = '5707806340';

    let currentSlide = 0;
    const totalSlides = 4;
    let autoSlideInterval;

    function goToSlide(index) {
      currentSlide = index;
      const wrapper = document.getElementById('slidesWrapper');
      wrapper.style.transform = `translateX(-${currentSlide * 100}vw)`;
      
      const dots = document.querySelectorAll('.dot');
      dots.forEach((dot, i) => {
        dot.classList.toggle('active', i === currentSlide);
      });

      resetAutoSlide();
    }

    function nextSlide() {
      currentSlide = (currentSlide + 1) % totalSlides;
      goToSlide(currentSlide);
    }

    function startAutoSlide() {
      autoSlideInterval = setInterval(nextSlide, 7000);
    }

    function resetAutoSlide() {
      clearInterval(autoSlideInterval);
      startAutoSlide();
    }

    window.addEventListener('DOMContentLoaded', () => {
      const urlParams = new URLSearchParams(window.location.search);
      const name = urlParams.get('name');
      if (name) {
        document.getElementById('guestNameDisplay').innerText = name;
        document.getElementById('wishSenderName').value = name;
      }
    });

    const audio = document.getElementById('bgMusic');
    const musicBtn = document.getElementById('musicToggle');

    function startExperience() {
      const overlay = document.getElementById('welcome-overlay');
      overlay.style.opacity = '0';
      setTimeout(() => {
        overlay.style.visibility = 'hidden';
      }, 800);

      try {
        let playPromise = audio.play();
        if (playPromise !== undefined) {
          playPromise.then(() => {
            musicBtn.classList.add('playing');
          }).catch(() => {
            console.log("Audio bloking");
          });
        }
      } catch (e) {}

      startAutoSlide();
    }

    function toggleMusic() {
      if (audio.paused) {
        audio.play().then(() => {
          musicBtn.classList.add('playing');
        }).catch(() => {});
      } else {
        audio.pause();
        musicBtn.classList.remove('playing');
      }
    }

    function sendToTelegram(text) {
      const url = `https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage`;
      return fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          chat_id: TELEGRAM_CHAT_ID,
          text: text,
          parse_mode: 'HTML'
        })
      });
    }

    function sendRSVP(status) {
      const guestName = document.getElementById('guestNameDisplay').innerText;
      const text = `<b>💌 TO'YGA TASHRIF JAVOBI</b>\n\n<b>Mehmon:</b> ${guestName}\n<b>Javob:</b> ${status}`;
      
      sendToTelegram(text).then(() => {
        alert("Rahmat! Javobingiz yetkazildi.");
      }).catch(() => {
        alert("Rahmat! Javobingiz yetkazildi.");
      });
    }

    function sendWish(e) {
      e.preventDefault();
      const sender = document.getElementById('wishSenderName').value;
      const msg = document.getElementById('wishMessage').value;
      const statusMsg = document.getElementById('statusMsg');

      const text = `<b>🎉 YANGI TABRIKNOMA!</b>\n\n<b>Yuboruvchi:</b> ${sender}\n<b>Tabrik:</b> ${msg}\n\n<i>Kamronbek & Nmalumxon Nikoh To'yi</i>`;

      statusMsg.innerText = "Yuborilmoqda...";

      sendToTelegram(text).then(() => {
        statusMsg.innerText = "Ezguli tilagingiz yetkazildi! Rahmat 💖";
        document.getElementById('wishMessage').value = '';
      }).catch(() => {
        statusMsg.innerText = "Ezguli tilagingiz yetkazildi! Rahmat 💖";
      });
    }
