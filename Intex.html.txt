  <script>
    // Firebase sozlamalari
    const firebaseConfig = {
      databaseURL: "https://taklifnoma-1-default-rtdb.firebaseio.com/"
    };

    firebase.initializeApp(firebaseConfig);
    const database = firebase.database();
    const wishesRef = database.ref('wishes');

    const music = document.getElementById('bgMusic');
    const musicBtn = document.getElementById('musicBtn');
    const musicStatus = document.getElementById('musicStatus');

    // Taklifnomani ochish
    function enterInvitation() {
      music.play().then(() => {
        musicStatus.innerText = "Yoniq";
      }).catch(err => console.log("Musiqa xatoligi:", err));

      musicBtn.style.display = "block";
      showPage('page1');
    }

    // Oynalarni almashtirish
    function showPage(pageId) {
      document.querySelectorAll('.page').forEach(page => {
        page.classList.remove('active');
      });
      document.getElementById(pageId).classList.add('active');
    }

    // Musiqa boshqaruvi
    function toggleMusic() {
      if (music.paused) {
        music.play();
        musicStatus.innerText = "Yoniq";
      } else {
        music.pause();
        musicStatus.innerText = "O'chirilgan";
      }
    }

    // Tabriklarni bazaga saqlash
    function addWish(event) {
      event.preventDefault();
      
      const nameInput = document.getElementById('guestName');
      const messageInput = document.getElementById('guestMessage');

      if(nameInput.value.trim() && messageInput.value.trim()) {
        const newWishRef = wishesRef.push();
        newWishRef.set({
          name: nameInput.value.trim(),
          message: messageInput.value.trim(),
          timestamp: Date.now()
        });

        nameInput.value = '';
        messageInput.value = '';
      }
    }

    // Bazadan tabriklarni yuklash
    wishesRef.on('child_added', (snapshot) => {
      const wish = snapshot.val();
      const wishesList = document.getElementById('wishesList');

      const wishDiv = document.createElement('div');
      wishDiv.className = 'wish-item';
      wishDiv.innerHTML = `<strong>${escapeHtml(wish.name)}</strong>${escapeHtml(wish.message)}`;

      wishesList.prepend(wishDiv);
    });

    function escapeHtml(text) {
      return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
    }
