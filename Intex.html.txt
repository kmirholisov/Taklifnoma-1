    <script>
        let currentSlide = 0;
        const totalSlides = 4;
        let autoSlideInterval;

        function openInvitation() {
            // Musiqani ijro etish
            const music = document.getElementById('bg-music');
            music.play().catch(error => console.log("Musiqa yuklanmadi:", error));

            // Ekranni va Kartani ochish
            document.getElementById('welcome-screen').classList.add('hidden');
            document.getElementById('card').classList.add('active');
            document.getElementById('bg-img').style.transform = 'scale(1)';

            // Avto surilishni boshlash (Har 5 sekundda)
            startAutoSlide();
        }

        function goToSlide(index) {
            currentSlide = index;
            const wrapper = document.getElementById('slides-wrapper');
            const dots = document.querySelectorAll('.dot');

            // Slayderni surish
            wrapper.style.transform = `translateX(-${currentSlide * 25}%)`;

            // Active nuqtani yangilash
            dots.forEach((dot, i) => {
                dot.classList.toggle('active', i === currentSlide);
            });
        }

        function nextSlide() {
            currentSlide = (currentSlide + 1) % totalSlides;
            goToSlide(currentSlide);
        }

        function startAutoSlide() {
            autoSlideInterval = setInterval(nextSlide, 5000); // 5000 ms = 5 sekund
        }
    </script>
