    <script>
        let currentSlide = 0;
        const totalSlides = 4;
        let autoSlideInterval;

        window.addEventListener('DOMContentLoaded', () => {
            const urlParams = new URLSearchParams(window.location.search);
            const nameParam = urlParams.get('name');
            if (nameParam) {
                document.getElementById('guest-name-display').innerText = nameParam;
            }
        });

        function openInvitation() {
            // Ekranni berkitish va kartani ko'rsatish
            document.getElementById('welcome-screen').classList.add('hidden');
            document.getElementById('card').classList.add('active');
            document.getElementById('bg-img').style.transform = 'scale(1)';

            // Musiqani chaling (xatosiz)
            const music = document.getElementById('bg-music');
            if (music) {
                music.play().catch(function(e) { console.log("Music play blocked", e); });
            }

            // Slayder va Taymerni boshlash
            startAutoSlide();
            startCountdown();
        }

        function toggleAudio() {
            const music = document.getElementById('bg-music');
            const icon = document.getElementById('audio-toggle').querySelector('i');
            if (music.paused) {
                music.play();
                icon.className = "fa-solid fa-volume-high";
            } else {
                music.pause();
                icon.className = "fa-solid fa-volume-xmark";
            }
        }

        function goToSlide(index) {
            currentSlide = index;
            document.getElementById('slides-wrapper').style.transform = `translateX(-${currentSlide * 25}%)`;
            document.querySelectorAll('.dot').forEach((dot, i) => {
                dot.classList.toggle('active', i === currentSlide);
            });
        }

        function nextSlide() {
            currentSlide = (currentSlide + 1) % totalSlides;
            goToSlide(currentSlide);
        }

        function startAutoSlide() {
            if (!autoSlideInterval) {
                autoSlideInterval = setInterval(nextSlide, 5000);
            }
        }

        function startCountdown() {
            const weddingDate = new Date("Sep 28, 2026 18:00:00").getTime();
            setInterval(() => {
                const now = new Date().getTime();
                const distance = weddingDate - now;

                if (distance > 0) {
                    document.getElementById("days").innerText = Math.floor(distance / (1000 * 60 * 60 * 24));
                    document.getElementById("hours").innerText = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                    document.getElementById("minutes").innerText = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
                    document.getElementById("seconds").innerText = Math.floor((distance % (1000 * 60)) / 1000);
                }
            }, 1000);
        }

        function confirmRSVP(btn, status) {
            document.querySelectorAll('.rsvp-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            
            const urlParams = new URLSearchParams(window.location.search);
            const guestName = urlParams.get('name') || "Noma'lum mehmon";

            // BotFather'dan olgan ma'lumotlaringizni shu yerga qo'ying:
            const telegramBotToken = 'YOUR_BOT_TOKEN_HERE';
            const chatId = 'YOUR_CHAT_ID_HERE';
            
            if (telegramBotToken !== 'YOUR_BOT_TOKEN_HERE') {
                const message = `📩 *Yangi Javob!*\n👤 *Mehmon:* ${guestName}\n📊 *Status:* ${status}`;
                fetch(`https://api.telegram.org/bot${telegramBotToken}/sendMessage`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ chat_id: chatId, text: message, parse_mode: 'Markdown' })
                });
            }

            alert("Rahmat! Javobingiz qabul qilindi: " + status);
        }
    </script>
